cmake_minimum_required(VERSION 3.16)
project(garmin2osmand VERSION 0.97.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Print per-command elapsed times even when building outside the Python pipeline.
if(CMAKE_GENERATOR MATCHES "Ninja|Makefiles")
  list(PREPEND CMAKE_CXX_COMPILER_LAUNCHER "${CMAKE_COMMAND}" -E time)
  if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.21)
    list(PREPEND CMAKE_CXX_LINKER_LAUNCHER "${CMAKE_COMMAND}" -E time)
  endif()
endif()
# Optional acceleration. The Python implementation remains available on systems
# without an Expat runtime. No dependencies are downloaded during build.
find_package(EXPAT QUIET)
if(NOT EXPAT_FOUND)
  # Python-capable Linux installations commonly have the shared runtime but
  # omit development headers and the unversioned linker symlink.
  find_library(GARMIN_EXPAT_RUNTIME NAMES expat libexpat.so.1)
  if(GARMIN_EXPAT_RUNTIME)
    add_library(EXPAT::EXPAT UNKNOWN IMPORTED)
    set_target_properties(EXPAT::EXPAT PROPERTIES
      IMPORTED_LOCATION "${GARMIN_EXPAT_RUNTIME}"
      INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/src/vendor/expat")
    set(EXPAT_FOUND TRUE)
  endif()
endif()
if(EXPAT_FOUND)
  add_executable(native_postprocess src/native_postprocess.cpp)
  target_link_libraries(native_postprocess PRIVATE EXPAT::EXPAT)
else()
  add_custom_target(native_postprocess
    COMMAND "${CMAKE_COMMAND}" -E echo "Expat runtime unavailable, using Python processing")
endif()
option(GARMIN_POSTPROCESS_ONLY "Configure only the native XML processing tool" OFF)
if(GARMIN_POSTPROCESS_ONLY)
  enable_testing()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(EXPAT_FOUND AND Python3_Interpreter_FOUND)
    add_test(NAME native_backend_regressions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_native_backend.py)
  endif()
  return()
endif()
find_package(PNG REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_executable(garmin2osmand src/main.cpp)
add_executable(garmin2osm src/vector.cpp)
target_link_libraries(garmin2osm PRIVATE PNG::PNG SQLite::SQLite3 Threads::Threads)
target_link_libraries(garmin2osmand PRIVATE PNG::PNG SQLite::SQLite3 Threads::Threads)
if(MSVC)
 target_compile_options(garmin2osmand PRIVATE /W4 /EHsc)
else()
 target_compile_options(garmin2osmand PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_test(NAME help COMMAND garmin2osm --help)

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME legacy_native_vector_parser COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_vector.py $<TARGET_FILE:garmin2osm>)
  add_test(NAME v092_native_osm_translation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_v092.py)
endif()
if(Python3_Interpreter_FOUND)
  add_test(NAME v093_memory_and_mapping_regressions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_v093.py)
endif()

if(Python3_Interpreter_FOUND)
  add_test(NAME v095_cli_config_and_obf_orchestration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_v095.py)
  add_test(NAME native_registry_audit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_registry_audit.py)
  add_test(NAME per_type_poi_hashes COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_hash_config.py $<TARGET_FILE:garmin2osm>)
endif()


if(EXPAT_FOUND AND Python3_Interpreter_FOUND)
  add_test(NAME native_backend_regressions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_native_backend.py)
endif()
